### Structural Consequence Paradigm

Rather than relying on manual element-wise counting or inclusion-exclusion sieving, Oppermann's and Legendre's Conjectures are formulated as direct, structural consequences of three algebraic and geometric invariants:

1. **Pair Configuration Uniqueness**:
   Decomposes the pair structure of odd integers in the Oppermann interval $[n^2 - n, n^2 + n]$ into bilateral prime pairs ($P_p$), unilateral left-side prime pairs ($M_l$), unilateral right-side prime pairs ($M_r$), and bilateral composite pairs ($C_p$), preserving the exact algebraic identity $a = 2P_p + M_l + M_r$.

2. **Structural Compensation & Non-Degeneration**:
   Even under total prime-pair extinction ($P_p = 0$), the geometric exclusion grid forces a positive non-degeneration floor ($M_l > 0 \land M_r > 0$). The structure encapsulates explicit non-degeneration witnesses (`witness_left`, `witness_right`), resolving prime existence directly without relying on external bridge hypotheses.

3. **Reduction to Legendre's Conjecture**:
   The existence of a prime $q$ in the right half-interval ($n^2 < q < n^2 + n$) naturally guarantees prime existence in the broader Legendre interval ($n^2 < p < (n+1)^2$) via the strict integer bound $n^2 + n < (n+1)^2$.

    function checkSift(x, mode) {
      if (x < 2) return false;
      if (x === 2) return true;
      if (x % 2 === 0) return false;

      const r = Math.floor(Math.sqrt(x));
      let maxDivisor = r;
      if (mode === '3') maxDivisor = 3;
      else if (mode === '3_5') maxDivisor = 5;
      else if (mode === '3_5_7') maxDivisor = 7;
      else if (mode === 'all') maxDivisor = r;

      for (let i = 3; i <= r; i += 2) {
        if (i <= maxDivisor) {
          if (x % i === 0 && x !== i) return false;
        } else {
          break;
        }
      }
      return true;
    }

    function computeData(maxN, siftMode) {
      const ns = [];
      const aArr = [], bArr = [];
      const MlArr = [], MrArr = [];
      const PpArr = [], CpArr = [];
      const HpArr = [], MpArr = [];
      const a2Arr = [];
      const b2Arr = [];
      const abArr = [];

      for (let n = 2; n <= maxN; n++) {
        const center = n * n;
        let a = 0, b = 0;
        for (let x = center - n + 1; x < center + n; x++) {
          if (x === center) continue;
          if (x % 2 === 0) continue;
          if (checkSift(x, siftMode)) a++;
          else b++;
        }

        let Ml = 0, Mr = 0, Pp = 0, Cp = 0;
        for (let k = 1; k < n; k++) {
          const L = center - k;
          const R = center + k;
          if (L % 2 === 0 || R % 2 === 0) continue;

          const Lp = checkSift(L, siftMode);
          const Rp = checkSift(R, siftMode);

          if (Lp && Rp) Pp++;
          else if (Lp && !Rp) Ml++;
          else if (!Lp && Rp) Mr++;
          else if (!Lp && !Rp) Cp++;
        }

        const denom = (2 * a + 2 * b || 1);
        const denom2 = (a + b || 1);
        const a2 = (a * a) / denom;
        const b2 = (b * b) / denom;
        const ab = (a * b) / denom2;

        const Hp = Cp + Pp;
        const Mp = Ml + Mr;

        ns.push(n);
        aArr.push(a);
        bArr.push(b);
        PpArr.push(Pp);
        CpArr.push(Cp);
        HpArr.push(Hp);
        MpArr.push(Mp);
        MlArr.push(Ml);
        MrArr.push(Mr);
        a2Arr.push(a2);
        b2Arr.push(b2);
        abArr.push(ab);
      }

      return { ns, aArr, bArr, PpArr, CpArr, HpArr, MpArr, MlArr, MrArr, a2Arr, b2Arr, abArr };
    }
